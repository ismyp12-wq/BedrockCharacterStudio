  function v4Problems(){
    const problems=[];const names=sourceBoneNames();
    if(state.mode!=="merge"&&!names.length)problems.push({type:"warning",text:"ยังไม่พบรายชื่อ Bone จากโมเดล"});
    if(state.mode==="outfit"){
      const missing=ROLES.filter(r=>!state.mapping[r.id]);if(missing.length)problems.push({type:"error",text:`ชุดเต็มตัวยังไม่ได้จับคู่: ${missing.map(x=>x.label).join(", ")}`});
      const values=Object.values(state.mapping).filter(Boolean);if(new Set(values).size!==values.length)problems.push({type:"error",text:"Bone ชุดเต็มตัวถูกเลือกซ้ำ"});
    }
    const info=window.lyV4TextureInfo;if(state.mode!=="merge"&&info){if(info.width>2048||info.height>2048)problems.push({type:"warning",text:`Texture ${info.width}×${info.height} มีขนาดใหญ่มาก`});if(!info.powerOfTwo)problems.push({type:"warning",text:`Texture ${info.width}×${info.height} ไม่ใช่ขนาดกำลังสอง`});}
    if(state.mode==="faces"&&document.querySelectorAll(".expression-row").length>16)problems.push({type:"warning",text:"สีหน้ามากกว่า 16 แบบ เมนูในเกมอาจยาว"});
    return problems;
