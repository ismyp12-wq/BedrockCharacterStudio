  function loadThree(){
    return new Promise((resolve,reject)=>{
      if(window.THREE){THREERef=window.THREE;resolve();return;}
      const script=document.createElement("script");script.src="https://cdn.jsdelivr.net/npm/three@0.161.0/build/three.min.js";script.onload=()=>{THREERef=window.THREE;resolve();};script.onerror=()=>reject(new Error("โหลด Three.js ไม่สำเร็จ"));document.head.appendChild(script);

  function ensureScene(){
    if(renderer||!THREERef)return;
    const host=$id("v4CanvasWrap");if(!host)return;host.innerHTML="";
    scene=new THREERef.Scene();camera=new THREERef.PerspectiveCamera(38,1,.1,1000);camera.position.set(38,28,48);
    renderer=new THREERef.WebGLRenderer({antialias:true,alpha:true});renderer.setPixelRatio(Math.min(devicePixelRatio,2));renderer.outputColorSpace=THREERef.SRGBColorSpace;host.appendChild(renderer.domElement);
    const hemi=new THREERef.HemisphereLight(0xffffff,0x343245,2.4);scene.add(hemi);const dir=new THREERef.DirectionalLight(0xffffff,2.1);dir.position.set(25,40,30);scene.add(dir);
    const grid=new THREERef.GridHelper(64,32,0x4d4a65,0x2d2b3e);grid.position.y=0;scene.add(grid);
    modelRoot=new THREERef.Group();scene.add(modelRoot);
    const resize=()=>{const r=host.getBoundingClientRect();const w=Math.max(260,r.width),h=Math.max(310,r.height);renderer.setSize(w,h,false);camera.aspect=w/h;camera.updateProjectionMatrix();};new ResizeObserver(resize).observe(host);resize();
    const canvas=renderer.domElement;canvas.addEventListener("pointerdown",e=>{drag=true;lastX=e.clientX;lastY=e.clientY;canvas.setPointerCapture(e.pointerId)});canvas.addEventListener("pointerup",()=>drag=false);canvas.addEventListener("pointermove",e=>{if(!drag||!modelRoot)return;modelRoot.rotation.y+=(e.clientX-lastX)*.012;modelRoot.rotation.x+=(e.clientY-lastY)*.008;modelRoot.rotation.x=Math.max(-1.2,Math.min(1.2,modelRoot.rotation.x));lastX=e.clientX;lastY=e.clientY;});canvas.addEventListener("wheel",e=>{e.preventDefault();camera.position.multiplyScalar(e.deltaY>0?1.08:.92);camera.position.clampLength(18,140);},{passive:false});
    (function animate(){requestAnimationFrame(animate);renderer.render(scene,camera);})();
  }
